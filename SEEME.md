# правила/сведение
# заходим в директорию проекта
cd arch-install
# всасываем все изменения(текущей ветки)
git pull
# или же пишем название ветки 
git pull origin "название ветки"
# создаем свою ветку
git branch "название ветки"
# тепрь переходим в нашу ветку
git checkout "название ветки"
# добавляем все файлы (. = все файлы в папке)
git add .
# делаем коммит
git commit -m "описание того что добавили"
# пушим
git push -u origin "название ветки которую мы создали и в которую мы перешли"
