# правила/сведение
# заходим в дерикторию проекта
cd "путь к дериктории"
# всасываем все изменения(текущей ветки)
git pull
# создаем свою ветку
git branch "название ветки"
# тепрь переходим в нашу ветку
git switch "название ветки"
# добавляем все изменения в текущей дериктории
git add .
# делаем коммит
git commit -m "описание того что добавили"
# пушим
git push origin "название ветки которую мы создали и в которую мы перешли"
# создаем файл .gitignore в директории в которой мы работаем с нашим проектом и гитом(у меня это arch-install)
touch ~/arch-install/.gitignore
# заходим в редактор (редактируем этот файл)
helix ~/arch-install/.gitignore
# пишем путь к файлу который мы будем игнорировать в .gitignore
~/путь/к/файлу
# сохраняем и выходим из редактора
Esc :wq
